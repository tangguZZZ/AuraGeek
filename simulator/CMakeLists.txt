cmake_minimum_required(VERSION 3.24)
project(AuraGeekSimulator LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(LVGL_SOURCE_DIR "D:/LVGL-Dev/sources/lvgl-9.5.0" CACHE PATH "Pinned LVGL source")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS ON CACHE BOOL "" FORCE)
find_package(SDL2 CONFIG REQUIRED)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/lvgl")
target_link_libraries(lvgl PUBLIC SDL2::SDL2)
file(GLOB UI_SOURCES CONFIGURE_DEPENDS "../src/ui/AuraUi.cpp" "../src/ui/resources/*.c")
add_executable(aurageek_sim main.cpp ../src/services/StockSeed.cpp ${UI_SOURCES})
add_executable(aurageek_tests tests.cpp ../src/services/StockSeed.cpp)
target_include_directories(aurageek_tests PRIVATE ../src)
add_test(NAME core_math COMMAND aurageek_tests)
target_include_directories(aurageek_sim PRIVATE ../src)
target_compile_definitions(aurageek_sim PRIVATE SDL_MAIN_HANDLED AURAGEEK_SIMULATOR)
target_link_libraries(aurageek_sim PRIVATE lvgl lvgl_demos lvgl_examples SDL2::SDL2)
add_custom_command(TARGET aurageek_sim POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:aurageek_sim>)
get_filename_component(MINGW_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
foreach(RUNTIME_DLL libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
  add_custom_command(TARGET aurageek_sim POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MINGW_BIN}/${RUNTIME_DLL}" $<TARGET_FILE_DIR:aurageek_sim>)
endforeach()
